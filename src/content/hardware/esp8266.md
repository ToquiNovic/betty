Las placas basadas en **ESP8266** (NodeMCU v2/v3, Wemos D1 Mini) son ideales para nodos de bajo costo.

---

## 💻 Código Fuente de Ejemplo

```cpp
#include <ESP8266WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>

const char* ssid = "TU_WIFI_SSID";
const char* password = "TU_WIFI_PASSWORD";

const char* mqtt_server = "192.168.1.100";
const int mqtt_port = 1883;

const char* sensor_id = "tu-sensor-uuid";
const char* api_key = "betty_live_...";
const char* mqtt_topic = "betty/sensor/tu-sensor-uuid/data";

WiFiClient espClient;
PubSubClient client(espClient);

void setup_wifi() {
  delay(10);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Conectado!");
}

void reconnect() {
  while (!client.connected()) {
    if (client.connect(sensor_id, sensor_id, api_key)) {
      Serial.println("Conectado a Betty MQTT!");
    } else {
      delay(5000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  setup_wifi();
  client.setServer(mqtt_server, mqtt_port);
}

void loop() {
  if (!client.connected()) reconnect();
  client.loop();

  static unsigned long lastTime = 0;
  if (millis() - lastTime > 10000) {
    lastTime = millis();

    StaticJsonDocument<150> doc;
    doc["temperature"] = 25.4;
    doc["humidity"] = 48.2;
    doc["origin_type"] = "sensor";

    char buffer[200];
    serializeJson(doc, buffer);
    client.publish(mqtt_topic, buffer);
  }
}
```
